MODULE Module9.bas
Attribute VB_Name = "Module9"
Public Function Work_Days(BegDate As Variant, EndDate As Variant) As Integer

 Dim WholeWeeks As Variant
 Dim DateCnt As Variant
 Dim EndDays As Integer

 On Error GoTo Err_Work_Days

 BegDate = DateValue(BegDate)
 EndDate = DateValue(EndDate)
 WholeWeeks = DateDiff("w", BegDate, EndDate)
 DateCnt = DateAdd("ww", WholeWeeks, BegDate)
 EndDays = 0

 Do While DateCnt <= EndDate
 If Format(DateCnt, "ddd") <> "Sun" And _
 Format(DateCnt, "ddd") <> "Sat" Then
 EndDays = EndDays + 1
 End If
 DateCnt = DateAdd("d", 1, DateCnt)
 Loop

 Work_Days = WholeWeeks * 5 + EndDays

Exit Function

Err_Work_Days:

 ' If either BegDate or EndDate is Null, return a zero
 ' to indicate that no workdays passed between the two dates.

 If Err.Number = 94 Then
 Work_Days = 0
 Exit Function
 Else
' If some other error occurs, provide a message.
 MsgBox "Error"
 End If

End Function

Function AreaTuboRedondo(De, tw)

'    If Unidade = "Métrico (mm)" Then
'
'        AreaTuboRedondo = 0.25 * WorksheetFunction.Pi() * (De * De - (De - tw) * (De - tw))
'
'    ElseIf Unidade = "Imperial (in)" Then
'
'        De = 25.4 * De
'
'        tw = 25.4 * tw
'
'        AreaTuboRedondo = 0.25 * WorksheetFunction.Pi() * (De * De - (De - tw) * (De - tw))

'    End If
    
    AreaTuboRedondo = 0.25 * WorksheetFunction.Pi() * (De * De - (De - 2 * tw) * (De - 2 * tw))
    
End Function


Function AreaTuboRetangular(we, he, tw)

'    If Unidade = "Métrico (mm)" Then
'
'        AreaTuboRetangular = we * he - (we - tw) * (he - tw)

'
'        tw = 25.4 * tw
'
'        AreaTuboRetangular = we * he - (we - tw) * (he - tw)

'    End If
    
    AreaTuboRetangular = we * he - (we - 2 * tw) * (he - 2 * tw)
    
End Function

    x = InputBox("Password")

    If x = "plan" Then
    
    '1. Abre Planeamento
        
        Workbooks("Met2_Plan_Perfis").Sheets("Planeamento").Activate
        
        '2. Cria Matriz de Dados
        
        Dim matr_dado As Variant
        
        
        '3. Ajustar Folha de Planeamento

        Columns.EntireColumn.Hidden = False
        Rows.EntireRow.Hidden = False
        
        With ActiveSheet
            If .FilterMode Then
                .ShowAllData
            End If
        End With
        
        'Worksheets("Planeamento").Range("BD7:BE200000").ClearContents
        
        '4. Filtrar Datas Previstas
        
        Worksheets("Planeamento").Range("A7:CS200000" & nregistostotal).AutoFilter Field:=44, Criteria1:="<>" & ""
        
        '5. Percorrer Dados Planeados
        
        Worksheets("Planeamento").Range("A6").Select
        
        ActiveCell.End(xlDown).Select
        
        ulti_linh_filt = Split(ActiveCell.Address, "$")(2)
        
        ntot_linh_filt = Worksheets("Planeamento").Range("AR7:" & "AR" & CStr(ulti_linh_filt)).SpecialCells(xlCellTypeVisible).Count
        
        '6. Redimensiona e adiciona valores a Matriz de Dados
        
        ReDim matr_dado(ntot_linh_filt, 8)
        
        Worksheets("Planeamento").Range("A7").Select

        nlinhasfechadas = 1
        
        Do Until IsEmpty(ActiveCell)

            If ActiveCell.EntireRow.Hidden Then

            Else
                
                '6.1. Fecha semanas abertas com data atual maior que a prevista
                
                If ActiveCell.Offset(0, 57).Value = "X" And ActiveCell.Offset(0, 58).Value <> "X" And ActiveCell.Offset(0, 53).Value > ActiveCell.Offset(0, 43).Value Then
    
                    '6.1.1. Adiciona os valores de ID
                    matr_dado(nlinhasfechadas, 1) = ActiveCell.Value
                    '6.1.2. Adiciona os valores das Linhas do Planeamento
                    matr_dado(nlinhasfechadas, 2) = Split(ActiveCell.Address, "$")(2)
                    '6.1.3. Adiciona valores de Semana Atual
                    matr_dado(nlinhasfechadas, 3) = ActiveCell.Offset(0, 54).Value
                    '6.1.4. Adiciona valores de Ano Atual
                    matr_dado(nlinhasfechadas, 4) = ActiveCell.Offset(0, 55).Value
                    '6.1.5. Adiciona valores de Area Secao de Corte Fechamento
                    matr_dado(nlinhasfechadas, 5) = ActiveCell.Offset(0, 56).Value
                    '6.1.6. Adiciona valores de Semana Fechada
                    matr_dado(nlinhasfechadas, 6) = "X"
                    
                    '6.1.6. Marca fechamento da semana
                    ActiveCell.Offset(0, 58).Value = "X"
                
                    nlinhasfechadas = nlinhasfechadas + 1
                    
                End If
    
            End If
            
            ActiveCell.Offset(1, 0).Select

        Loop
        
        MsgBox (CStr(nlinhasfechadas - 1) & " novas linhas fechadas!")
        
        '7. Abre Historico de Planeamento

        Workbooks.Open ("I:\usr\dpm\dp\3_Planeamento\Plan_Cort.Trans_Perfis\Historico_Plan_Perfis.xlsx")
        Workbooks("Historico_Plan_Perfis.xlsx").Worksheets("HistPerfis").Activate

        '8. Percorre linhas preenchidas no Historico de Planeamento e atualiza linhas fechadas

        Worksheets("HistPerfis").Range("A2").Select

        nlinhas_hist = 1

        Do Until IsEmpty(ActiveCell)
            
            For i = 1 To UBound(matr_dado, 1)
            
                '8.1. Verifica a equivalencia dos IDs
            
                If ActiveCell.Value = matr_dado(i, 1) Then
                
                    '8.1.1. Adiciona valores de Area Secao de Corte Fechamento
                    ActiveCell.Offset(0, 6).Value = matr_dado(i, 5)
                    
                    '8.1.2. Adiciona valores da Semana de Fechamento
                    ActiveCell.Offset(0, 11).Value = matr_dado(i, 3)
                    
                    '8.1.3. Adiciona valores do Ano de Fechamento
                    ActiveCell.Offset(0, 12).Value = matr_dado(i, 4)
                    
                    '8.1.4. Calcula valores Area de Corte Efetiva
                    
                    On Error Resume Next
                    ActiveCell.Offset(0, 7).Value = ActiveCell.Offset(0, 6).Value - ActiveCell.Offset(0, 5).Value
                    
                    '8.1.5. Calcula valores Area de Corte Planeda e Nao Planeada
                    
                    On Error Resume Next
                    If ActiveCell.Offset(0, 7).Value >= ActiveCell.Offset(0, 4).Value Then
                    
                        On Error Resume Next
                        ActiveCell.Offset(0, 9).Value = ActiveCell.Offset(0, 4).Value
                    
                        On Error Resume Next
                        ActiveCell.Offset(0, 8).Value = ActiveCell.Offset(0, 7).Value - ActiveCell.Offset(0, 4).Value
                        
                        '8.1.5.1. Destaca linhas cumpridas
                        
                        matr_dado(i, 7) = ""
                        
                    Else
                    
                        On Error Resume Next
                        ActiveCell.Offset(0, 9).Value = ActiveCell.Offset(0, 7).Value
                        
                        ActiveCell.Offset(0, 8).Value = 0
                        
                        '8.1.5.2. Destaca linhas nao cumpridas
                        
                        matr_dado(i, 7) = "X"
                        
                    End If
                    
                    '8.1.6. Adiciona valores de Semana Fechada
                    ActiveCell.Offset(0, 14).Value = matr_dado(i, 6)
                    
                    '8.1.7. Calcula taxa de cumprimento
                    
                    On Error Resume Next
                    ActiveCell.Offset(0, 16).Value = ActiveCell.Offset(0, 9).Value / ActiveCell.Offset(0, 4).Value
                    matr_dado(i, 8) = ActiveCell.Offset(0, 16).Value
                    
                End If
            
            Next
            
            nlinhas_hist = nlinhas_hist + 1
            
            ActiveCell.Offset(1, 0).Select

        Loop
        
        '9. Fecha o Histórico
        
        MsgBox ("Registo de semanas fechadas no histórico concluído!")
        
        Workbooks("Historico_Plan_Perfis.xlsx").Worksheets("HistPerfis").Activate
        ActiveWorkbook.Close True
        
        '10. Reabre o Planeamento
        
        Workbooks("Met2_Plan_Perfis.xlsm").Worksheets("Planeamento").Activate
        
        '11. Busca linhas cumpridas
        
        Worksheets("Planeamento").Range("A7").Select
        
        Do Until IsEmpty(ActiveCell)

            If ActiveCell.EntireRow.Hidden Then

            Else
                
                If ActiveCell.Offset(0, 57).Value = "X" And ActiveCell.Offset(0, 58).Value = "X" Then
                    
                    For i = 1 To UBound(matr_dado, 1)
                    
                        If matr_dado(i, 1) = ActiveCell.Value Then
                        
                            ActiveCell.Offset(0, 59).Value = matr_dado(i, 8)
                            
                            If matr_dado(i, 7) <> "X" Then
                                                
                                'ActiveCell.Offset(0, 43).ClearContents
                            
                                'ActiveCell.Offset(0, 43).Value = "Feito"
                            
                            End If
                            
                        End If
                    
                    Next
                    
                End If
    
            End If
            
            ActiveCell.Offset(1, 0).Select

        Loop
